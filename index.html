1前往 Google Apps Script（需登入 Google）→ 點「新增專案」→ 刪除預設程式碼，貼入下方程式碼後按 Ctrl+S 儲存
2點右上角「部署」→「新增部署作業」→ 類型選 網頁應用程式 → 執行身分選 我 → 存取選 所有人 → 按「部署」→ 複製網頁應用程式網址
3將網址貼入下方，按「上傳」即可同步。將同一網址分享給對方填入，雙方即可共用同一份資料。
貼入 Google Apps Script 的程式碼：
const SHEET_NAME = 'ApplianceData'; function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); return ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); } function doGet() { const v = getSheet().getRange('A1').getValue(); return ContentService.createTextOutput(v || '{}') .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { getSheet().getRange('A1').setValue(e.postData.contents); getSheet().getRange('B1').setValue(new Date().toLocaleString('zh-TW')); return ContentService.createTextOutput('{"ok":true}') .setMimeType(ContentService.MimeType.JSON); }
